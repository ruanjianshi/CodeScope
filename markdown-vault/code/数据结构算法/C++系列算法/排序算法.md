// 1 冒泡排序：相邻两两比较，大的往后冒（这是参考实现）
void bubbleSort(Array &a) {
  for (size_t i = 0; i + 1 < a.size(); ++i) {
    for (size_t j = 0; j + 1 < a.size() - i; ++j) {
      if (a[j] > a[j + 1])
        std::swap(a[j], a[j + 1]);
    }
  }
}

    let swapped = false;
    for (let j = 0; j < n - 1 - i; j++) {
      yield { compare: [j, j+1], sorted: sorted.slice(), line: 4, type: 'compare' };
      if (bars[j].value > bars[j+1].value) {

function* mergeRec(bars, l, r, sorted) {
  if (l >= r) return;
  const m = Math.floor((l + r) / 2);
  yield* mergeRec(bars, l, m, sorted);
  yield* mergeRec(bars, m + 1, r, sorted);

function* mergeDo(bars, l, m, r, sorted) {
  const L = bars.slice(l, m + 1); const R = bars.slice(m + 1, r + 1);
  let i = 0, j = 0, k = l;
  while (i < L.length && j < R.length) {
    yield { compare: [l+i, m+1+j], sorted: sorted.slice(), line: 7, type: 'compare' };
    if (L[i].value <= R[j].value) bars[k] = L[i++]; else bars[k] = R[j++];
    yield { swap: [k], sorted: sorted.slice(), line: 8, type: 'overwrite' };
    k++;
  }
  while (i < L.length) { bars[k] = L[i++]; yield { swap: [k], sorted: sorted.slice(), line: 10, type: 'overwrite' }; k++; }
  while (j < R.length) { bars[k] = R[j++]; yield { swap: [k], sorted: sorted.slice(), line: 11, type: 'overwrite' }; k++; }
}

function* quickRec(bars, lo, hi, sorted) {
  if (lo > hi) return;
  if (lo === hi) { sorted[lo] = true; return; }
  const pivotVal = bars[hi].value;
  let i = lo - 1;
  for (let j = lo; j < hi; j++) {
    yield { compare: [j], pivot: hi, sorted: sorted.slice(), line: 3, type: 'compare' };
    if (bars[j].value <= pivotVal) {
      i++;
      if (i !== j) {
        [bars[i], bars[j]] = [bars[j], bars[i]];
        yield { swap: [i, j], pivot: hi, sorted: sorted.slice(), line: 5, type: 'swap' };
      }
    }
  }
  [bars[i + 1], bars[hi]] = [bars[hi], bars[i + 1]];
  yield { swap: [i+1, hi], pivot: i+1, sorted: sorted.slice(), line: 8, type: 'swap' };
  sorted[i + 1] = true;
  yield { pivot: i+1, sorted: sorted.slice(), line: 9, type: 'mark' };
  yield* quickRec(bars, lo, i, sorted);
  yield* quickRec(bars, i + 2, hi, sorted);
}

function* heapSort(bars) {
  const n = bars.length; const sorted = new Array(n).fill(false);
  yield { line: 12, sorted: sorted.slice(), type: 'idle' };
  for (let i = Math.floor(n/2) - 1; i >= 0; i--) yield* siftDown(bars, i, n, sorted);
  for (let i = n - 1; i > 0; i--) {
    [bars[0], bars[i]] = [bars[i], bars[0]];
    yield { swap: [0, i], sorted: sorted.slice(), line: 16, type: 'swap' };
    sorted[i] = true;
    yield* siftDown(bars, 0, i, sorted);
  }
  for (let i = 0; i < n; i++) sorted[i] = true;
  yield { sorted: sorted.slice(), line: 19, type: 'done' };
}
function* siftDown(bars, i, n, sorted) {
  while (true) {
    let largest = i; const l = 2 * i + 1, r = 2 * i + 2;
    if (l < n) {
      yield { compare: [largest, l], sorted: sorted.slice(), line: 4, type: 'compare' };
      if (bars[l].value > bars[largest].value) largest = l;
    }
    if (r < n) {
      yield { compare: [largest, r], sorted: sorted.slice(), line: 5, type: 'compare' };
      if (bars[r].value > bars[largest].value) largest = r;
    }
    if (largest === i) break;
    [bars[i], bars[largest]] = [bars[largest], bars[i]];
    yield { swap: [i, largest], sorted: sorted.slice(), line: 7, type: 'swap' };
    i = largest;
  }
}

function* countingSort(bars) {

function* bucketSort(bars) {

    buckets[bi].push(bars[i]);

  const arr = [];

    bars[i] = arr[i];
    yield { swap: [i], sorted: sorted.slice(), line: 13, type: 'overwrite' };

// 链表冒泡排序
function* bubbleSortLL(nodes) {
  const n = nodes.length;
  const sorted = new Array(n).fill(false);
  yield { line: 2, sorted: sorted.slice(), type: 'idle' };
  let swapped = true;
  let lptr = -1;   // 已排序边界
  while (swapped) {
    yield { line: 4, sorted: sorted.slice(), type: 'mark' };
    swapped = false;
    yield { line: 5, sorted: sorted.slice(), type: 'mark' };
    for (let j = 0; j < n - 1 - (lptr + 1); j++) {
      yield { compare: [j, j+1], sorted: sorted.slice(), line: 8, type: 'compare' };
      if (nodes[j].value > nodes[j+1].value) {
        [nodes[j].value, nodes[j+1].value] = [nodes[j+1].value, nodes[j].value];
        yield { swap: [j, j+1], sorted: sorted.slice(), line: 9, type: 'swap' };
        swapped = true;
      }
    }
    lptr++;
    sorted[n - 1 - lptr] = true;
    yield { line: 14, sorted: sorted.slice(), type: 'mark' };
  }
  for (let i = 0; i < n; i++) sorted[i] = true;
  yield { sorted: sorted.slice(), line: 16, type: 'done' };
}

// 链表归并排序（用数组模拟归并过程）
function* mergeSortLL(nodes) {
  const n = nodes.length;
  const sorted = new Array(n).fill(false);
  yield { line: 14, sorted: sorted.slice(), type: 'idle' };
  yield* mergeSortLLRec(nodes, 0, n - 1, sorted);
  for (let i = 0; i < n; i++) sorted[i] = true;
  yield { sorted: sorted.slice(), line: 15, type: 'done' };
}
function* mergeSortLLRec(nodes, l, r, sorted) {
  if (l >= r) return;
  const m = Math.floor((l + r) / 2);
  yield* mergeSortLLRec(nodes, l, m, sorted);
  yield* mergeSortLLRec(nodes, m + 1, r, sorted);
  yield* mergeSortLLDo(nodes, l, m, r, sorted);
}
function* mergeSortLLDo(nodes, l, m, r, sorted) {
  // 复制出左右两段的值
  const Lvals = nodes.slice(l, m + 1).map(x => x.value);
  const Rvals = nodes.slice(m + 1, r + 1).map(x => x.value);
  let i = 0, j = 0, k = l;
  yield { line: 6, sorted: sorted.slice(), type: 'mark' };
  while (i < Lvals.length && j < Rvals.length) {
    yield { compare: [l + i, m + 1 + j], sorted: sorted.slice(), line: 7, type: 'compare' };
    if (Lvals[i] <= Rvals[j]) {
      nodes[k].value = Lvals[i++];
    } else {
      nodes[k].value = Rvals[j++];
    }
    yield { swap: [k], sorted: sorted.slice(), line: 7, type: 'overwrite' };
    k++;
  }
  while (i < Lvals.length) {
    nodes[k].value = Lvals[i++];
    yield { swap: [k], sorted: sorted.slice(), line: 8, type: 'overwrite' };
    k++;
  }
  while (j < Rvals.length) {
    nodes[k].value = Rvals[j++];
    yield { swap: [k], sorted: sorted.slice(), line: 8, type: 'overwrite' };
    k++;
  }
}

const CPP_KEYWORDS = ['void','int','bool','char','float','double','if','else','for','while',
  'return','break','continue','true','false','const','new','delete','auto','long','short',
  'unsigned','signed','struct','class','public','private','protected','namespace','using',
  'template','typename','static','inline','virtual','size_t','nullptr'];
const KEYWORD_RE = new RegExp('\\b(' + CPP_KEYWORDS.join('|') + ')\\b', 'g');

function highlightCpp(text) {
  if (text === '') return '&nbsp;';
  let s = text.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');
  const stash = [];
  const TOKEN = '\uE000';
  s = s.replace(/(\/\/.*$)/g, m => { stash.push('<span class="cmt">'+m+'</span>'); return TOKEN+(stash.length-1)+TOKEN; });
  s = s.replace(/("[^"]*")/g, m => { stash.push('<span class="str">'+m+'</span>'); return TOKEN+(stash.length-1)+TOKEN; });
  s = s.replace(KEYWORD_RE, '<span class="kw">$1</span>');
  s = s.replace(/\bstd::(\w+)/g, '<span class="fn">std::$1</span>');
  s = s.replace(/\b(\d+)\b/g, '<span class="num">$1</span>');
  s = s.replace(new RegExp(TOKEN+'(\\d+)'+TOKEN, 'g'), (_, i) => stash[+i]);
  return s;
}

function updateTiming() {
  const s = parseInt($('speed').value);
  const delay = Math.max(30, Math.round(520 - s * 4.9));

/* ============ 数组布局 ============ */
function layoutBars() {
  const W = barsEl.clientWidth;
  const H = barsEl.clientHeight;
  const n = bars.length;
  if (n === 0 || W === 0 || H === 0) return;
  const margin = 6;
  const usable = W - margin * 2;
  const gap = Math.max(1, Math.min(8, 80 / n));
  const barW = (usable - gap * (n - 1)) / n;
  const maxV = Math.max(...bars.map(b => b.value));
  const maxH = H;
  bars.forEach((bar, i) => {
    const x = margin + i * (barW + gap);
    bar.el.style.left = x + 'px';
    bar.el.style.width = barW + 'px';
    const h = Math.max(4, (bar.value / maxV) * maxH);
    bar.el.style.height = h + 'px';
    bar.el.textContent = n <= 30 ? bar.value : '';
  });
}

/* ============ 链表布局 ============ */
function layoutNodes() {
  const W = nodesEl.clientWidth;
  const n = nodes.length;
  if (n === 0 || W === 0) return;
  // 根据节点数动态计算节点尺寸
  const gap = 2;
  const arrowW = 22;
  const totalArrows = (n - 1) * arrowW;
  const usable = W - 40 - totalArrows - (n - 1) * gap * 2;
  let boxW = Math.floor(usable / n);
  boxW = Math.max(30, Math.min(64, boxW));
  const boxH = boxW;
  nodes.forEach(node => {
    node.el.box.style.width = boxW + 'px';
    node.el.box.style.height = boxH + 'px';
    node.el.val.style.fontSize = Math.max(9, Math.min(14, Math.floor(boxW / 4))) + 'px';
  });
}

/* ============ 创建视图 ============ */
function createBars() {
  barsEl.innerHTML = '';
  bars = [];
  for (let i = 0; i < dataSize; i++) {
    const v = Math.floor(Math.random() * 92) + 8;
    const el = document.createElement('div');
    el.className = 'bar';
    barsEl.appendChild(el);
    bars.push({ id: i, value: v, el });
  }
  requestAnimationFrame(layoutBars);
}

function createNodes() {
  nodesEl.innerHTML = '';
  nodes = [];
  for (let i = 0; i < dataSize; i++) {
    const v = Math.floor(Math.random() * 92) + 8;
    const wrapper = document.createElement('div');
    wrapper.className = 'node-wrapper';
    const box = document.createElement('div');
    box.className = 'node-box';
    const val = document.createElement('div');
    val.className = 'node-val';
    val.textContent = v;
    const ptr = document.createElement('div');
    ptr.className = 'node-ptr';
    ptr.textContent = 'next';
    box.appendChild(val);
    box.appendChild(ptr);
    wrapper.appendChild(box);
    // 除了最后一个节点，都加一个箭头
    if (i < dataSize - 1) {
      const arrow = document.createElement('div');
      arrow.className = 'node-arrow';
      arrow.textContent = '→';
      wrapper.appendChild(arrow);
      nodes.push({ id: i, value: v, el: { wrapper, box, val, ptr, arrow } });
    } else {
      // 最后一个节点，用特殊标记
      wrapper.classList.add('null-tail');
      const arrow = document.createElement('div');
      arrow.className = 'node-arrow';
      arrow.textContent = '→';
      wrapper.appendChild(arrow);
      nodes.push({ id: i, value: v, el: { wrapper, box, val, ptr, arrow } });
    }
    nodesEl.appendChild(wrapper);
  }
  requestAnimationFrame(layoutNodes);
}

/* ============ 应用数组状态 ============ */
function applyArrayState(state) {
  bars.forEach(bar => bar.el.classList.remove('cmp', 'swap', 'piv', 'srt'));
  if (state.sorted) state.sorted.forEach((isS, i) => { if (isS && bars[i]) bars[i].el.classList.add('srt'); });
  if (state.pivot !== undefined && state.pivot >= 0 && bars[state.pivot]) bars[state.pivot].el.classList.add('piv');
  if (state.compare) state.compare.forEach(i => { if (bars[i]) bars[i].el.classList.add('cmp'); });
  if (state.swap) state.swap.forEach(i => { if (bars[i]) bars[i].el.classList.add('swap'); });
  layoutBars();
  if (state.line !== undefined && state.line >= 0) highlightLine(state.line);
}

/* ============ 应用链表状态 ============ */
function applyLinkedListState(state) {
  nodes.forEach(node => {
    node.el.box.classList.remove('cmp', 'swap', 'piv', 'srt', 'cur');
    if (node.el.arrow) node.el.arrow.classList.remove('active');
  });
  if (state.sorted) state.sorted.forEach((isS, i) => { if (isS && nodes[i]) nodes[i].el.box.classList.add('srt'); });
  if (state.pivot !== undefined && state.pivot >= 0 && nodes[state.pivot]) nodes[state.pivot].el.box.classList.add('piv');
  if (state.compare) state.compare.forEach(i => {
    if (nodes[i]) {
      nodes[i].el.box.classList.add('cmp');
      if (nodes[i].el.arrow) nodes[i].el.arrow.classList.add('active');
    }
  });
  if (state.swap) state.swap.forEach(i => { if (nodes[i]) nodes[i].el.box.classList.add('swap'); });
  // 更新节点数值（可能被交换或覆盖）
  nodes.forEach((node, i) => {
    if (node.el.val.textContent !== String(node.value)) {
      node.el.val.textContent = node.value;
    }
  });
  if (state.line !== undefined && state.line >= 0) highlightLine(state.line);
}

  stopTimer();

function ensureGenerator() {
  if (!generator) {
    const algo = getAlgoTable()[currentAlgo];
    generator = viewMode === 'array' ? algo.gen(bars) : algo.gen(nodes);
    stats = { compare: 0, swap: 0, step: 0 };
    updateStats();
  }
}

function applyState(state) {
  if (viewMode === 'array') applyArrayState(state);
  else applyLinkedListState(state);
}

function loop() {
  if (!playing) return;
  ensureGenerator();
  const { value, done } = generator.next();
  if (done) {
    playing = false;
    $('playBtn').textContent = '▶ 播放';
    $('playBtn').classList.add('pri');
    $('statusLabel').textContent = '已完成 ✓';
    return;
  }
  countStats(value);
  applyState(value);
  updateStats();
  $('statusLabel').textContent = '执行第 ' + stats.step + ' 步';
  timer = setTimeout(loop, updateTiming());
}

#include "Sort.h"

// ============================================================
//  简单测试：用几组数据跑一遍十种排序，打印结果和是否升序
//  运行这个片段就行（CodeScope 会把同目录的 .cpp 一起编译）
// ============================================================

int passed = 0, total = 0;

// 用一个函数指针跑一种排序，并检查结果是否升序
void test(const char* name, void (*sortFn)(Array&), Array data) {
  const size_t n = data.size();
  sortFn(data);
  const bool ok = std::is_sorted(data.begin(), data.end()) && data.size() == n;
  ++total;
  if (ok) ++passed;
  std::cout << (ok ? "  ✅ " : "  ❌ ") << name << "  ";
  for (int v : data) std::cout << v << ' ';
  std::cout << "\n";
}

// 一组数据 → 跑全部十种排序
void run(const char* title, Array data) {
  std::cout << "\n【" << title << "】原始数据: ";
  for (int v : data) std::cout << v << ' ';
  std::cout << "\n";
  test("冒泡排序", bubbleSort, data);
  test("选择排序", selectionSort, data);
  test("插入排序", insertionSort, data);
  test("希尔排序", shellSort, data);
  test("归并排序", mergeSort, data);
  test("快速排序", quickSort, data);
  test("堆排序  ", heapSort, data);
  test("计数排序", countingSort, data);
  test("桶排序  ", bucketSort, data);
  test("基数排序", radixSort, data);
}

int main() {
  run("普通乱序", { 5, 2, 8, 1, 9, 3, 7 });
  run("完全逆序", { 9, 8, 7, 6, 5, 4, 3, 2, 1 });
  run("重复 + 负数", { 3, -1, 3, 0, -5, 3, -1 });

  std::cout << "\n通过 " << passed << " / " << total << "\n";
  return passed == total ? 0 : 1;
}
