/* ============================================================
   1. 算法生成器
   每个 yield 返回一个"步骤快照"，包含：
   - line:     要高亮的代码行
   - compare:  正在比较的柱子索引数组
   - swap:     正在交换/写入的柱子索引数组（同时触发位置重排）
   - pivot:    基准柱子索引
   - sorted:   已排序布尔数组
   - type:     步骤类型：'compare' | 'swap' | 'overwrite' | 'done'
   ============================================================ */

function* bubbleSort(bars) {
  const n = bars.length;
  const sorted = new Array(n).fill(false);
  yield { line: 1, sorted: sorted.slice(), type: 'idle' };
  for (let i = 0; i < n - 1; i++) {
    let swapped = false;
    for (let j = 0; j < n - 1 - i; j++) {
      yield { compare: [j, j+1], sorted: sorted.slice(), line: 4, type: 'compare' };
      if (bars[j].value > bars[j+1].value) {
        [bars[j], bars[j+1]] = [bars[j+1], bars[j]];
        yield { swap: [j, j+1], sorted: sorted.slice(), line: 5, type: 'swap' };
        swapped = true;
      }
    }
    sorted[n - 1 - i] = true;
    yield { sorted: sorted.slice(), line: 9, type: 'mark' };
    if (!swapped) break;
  }
  for (let i = 0; i < n; i++) sorted[i] = true;
  yield { sorted: sorted.slice(), line: 11, type: 'done' };
}

function* selectionSort(bars) {
  const n = bars.length;
  const sorted = new Array(n).fill(false);
  yield { line: 1, sorted: sorted.slice(), type: 'idle' };
  for (let i = 0; i < n - 1; i++) {
    let minIdx = i;
    yield { compare: [i], sorted: sorted.slice(), line: 3, type: 'compare' };
    for (let j = i + 1; j < n; j++) {
      yield { compare: [j, minIdx], sorted: sorted.slice(), line: 4, type: 'compare' };
      if (bars[j].value < bars[minIdx].value) minIdx = j;
    }
    if (minIdx !== i) {
      [bars[i], bars[minIdx]] = [bars[minIdx], bars[i]];
      yield { swap: [i, minIdx], sorted: sorted.slice(), line: 8, type: 'swap' };
    }
    sorted[i] = true;
  }
  sorted[n - 1] = true;
  yield { sorted: sorted.slice(), line: 10, type: 'done' };
}

function* insertionSort(bars) {
  const n = bars.length;
  const sorted = new Array(n).fill(false);
  sorted[0] = true;
  yield { line: 1, sorted: sorted.slice(), type: 'idle' };
  for (let i = 1; i < n; i++) {
    yield { compare: [i], sorted: sorted.slice(), line: 2, type: 'compare' };
    let j = i;
    while (j > 0 && bars[j-1].value > bars[j].value) {
      yield { compare: [j-1, j], sorted: sorted.slice(), line: 4, type: 'compare' };
      [bars[j-1], bars[j]] = [bars[j], bars[j-1]];
      yield { swap: [j-1, j], sorted: sorted.slice(), line: 5, type: 'swap' };
      j--;
    }
    sorted[i] = true;
  }
  for (let i = 0; i < n; i++) sorted[i] = true;
  yield { sorted: sorted.slice(), line: 10, type: 'done' };
}

function* shellSort(bars) {
  const n = bars.length;
  const sorted = new Array(n).fill(false);
  yield { line: 1, sorted: sorted.slice(), type: 'idle' };
  for (let gap = Math.floor(n/2); gap > 0; gap = Math.floor(gap/2)) {
    for (let i = gap; i < n; i++) {
      let j = i;
      yield { compare: [i], sorted: sorted.slice(), line: 3, type: 'compare' };
      while (j >= gap && bars[j-gap].value > bars[j].value) {
        yield { compare: [j-gap, j], sorted: sorted.slice(), line: 5, type: 'compare' };
        [bars[j-gap], bars[j]] = [bars[j], bars[j-gap]];
        yield { swap: [j-gap, j], sorted: sorted.slice(), line: 6, type: 'swap' };
        j -= gap;
      }
    }
  }
  for (let i = 0; i < n; i++) sorted[i] = true;
  yield { sorted: sorted.slice(), line: 12, type: 'done' };
}

function* mergeSort(bars) {
  const n = bars.length;
  const sorted = new Array(n).fill(false);
  yield { line: 15, sorted: sorted.slice(), type: 'idle' };
  yield* mergeRec(bars, 0, n - 1, sorted);
  for (let i = 0; i < n; i++) sorted[i] = true;
  yield { sorted: sorted.slice(), line: 20, type: 'done' };
}
function* mergeRec(bars, l, r, sorted) {
  if (l >= r) return;
  const m = Math.floor((l + r) / 2);
  yield* mergeRec(bars, l, m, sorted);
  yield* mergeRec(bars, m + 1, r, sorted);
  yield* mergeDo(bars, l, m, r, sorted);
}
function* mergeDo(bars, l, m, r, sorted) {
  const L = bars.slice(l, m + 1);
  const R = bars.slice(m + 1, r + 1);
  let i = 0, j = 0, k = l;
  while (i < L.length && j < R.length) {
    yield { compare: [l+i, m+1+j], sorted: sorted.slice(), line: 7, type: 'compare' };
    if (L[i].value <= R[j].value) bars[k] = L[i++];
    else bars[k] = R[j++];
    yield { swap: [k], sorted: sorted.slice(), line: 8, type: 'overwrite' };
    k++;
  }
  while (i < L.length) { bars[k] = L[i++]; yield { swap: [k], sorted: sorted.slice(), line: 10, type: 'overwrite' }; k++; }
  while (j < R.length) { bars[k] = R[j++]; yield { swap: [k], sorted: sorted.slice(), line: 11, type: 'overwrite' }; k++; }
}

function* quickSort(bars) {
  const n = bars.length;
  const sorted = new Array(n).fill(false);
  yield { line: 14, sorted: sorted.slice(), type: 'idle' };
  yield* quickRec(bars, 0, n - 1, sorted);
  for (let i = 0; i < n; i++) sorted[i] = true;
  yield { sorted: sorted.slice(), line: 17, type: 'done' };
}
function* quickRec(bars, lo, hi, sorted) {
  if (lo > hi) return;
  if (lo === hi) { sorted[lo] = true; return; }
  const pivotVal = bars[hi].value;
  let i = lo - 1;
  for (let j = lo; j < hi; j++) {
    yield { compare: [j], pivot: hi, sorted: sorted.slice(), line: 3, type: 'compare' };
    if (bars[j].value <= pivotVal) {
      i++;
      if (i !== j) {
        [bars[i], bars[j]] = [bars[j], bars[i]];
        yield { swap: [i, j], pivot: hi, sorted: sorted.slice(), line: 5, type: 'swap' };
      }
    }
  }
  [bars[i + 1], bars[hi]] = [bars[hi], bars[i + 1]];
  yield { swap: [i+1, hi], pivot: i+1, sorted: sorted.slice(), line: 8, type: 'swap' };
  sorted[i + 1] = true;
  yield { pivot: i+1, sorted: sorted.slice(), line: 9, type: 'mark' };
  yield* quickRec(bars, lo, i, sorted);
  yield* quickRec(bars, i + 2, hi, sorted);
}

function* heapSort(bars) {
  const n = bars.length;
  const sorted = new Array(n).fill(false);
  yield { line: 12, sorted: sorted.slice(), type: 'idle' };
  for (let i = Math.floor(n/2) - 1; i >= 0; i--) {
    yield* siftDown(bars, i, n, sorted);
  }
  for (let i = n - 1; i > 0; i--) {
    [bars[0], bars[i]] = [bars[i], bars[0]];
    yield { swap: [0, i], sorted: sorted.slice(), line: 16, type: 'swap' };
    sorted[i] = true;
    yield* siftDown(bars, 0, i, sorted);
  }
  for (let i = 0; i < n; i++) sorted[i] = true;
  yield { sorted: sorted.slice(), line: 19, type: 'done' };
}
function* siftDown(bars, i, n, sorted) {
  while (true) {
    let largest = i;
    const l = 2 * i + 1, r = 2 * i + 2;
    if (l < n) {
      yield { compare: [largest, l], sorted: sorted.slice(), line: 4, type: 'compare' };
      if (bars[l].value > bars[largest].value) largest = l;
    }
    if (r < n) {
      yield { compare: [largest, r], sorted: sorted.slice(), line: 5, type: 'compare' };
      if (bars[r].value > bars[largest].value) largest = r;
    }
    if (largest === i) break;
    [bars[i], bars[largest]] = [bars[largest], bars[i]];
    yield { swap: [i, largest], sorted: sorted.slice(), line: 7, type: 'swap' };
    i = largest;
  }
}

function* countingSort(bars) {
  const n = bars.length;
  const sorted = new Array(n).fill(false);
  const maxV = Math.max(...bars.map(b => b.value));
  const minV = Math.min(...bars.map(b => b.value));
  const count = new Array(maxV - minV + 1).fill(0);
  yield { line: 4, sorted: sorted.slice(), type: 'idle' };
  for (let i = 0; i < n; i++) {
    count[bars[i].value - minV]++;
    yield { compare: [i], sorted: sorted.slice(), line: 4, type: 'compare' };
  }
  const original = bars.slice();
  const arr = [];
  for (let v = 0; v < count.length; v++) {
    for (let c = 0; c < count[v]; c++) {
      const found = original.find(b => b.value === v + minV && !arr.includes(b));
      if (found) arr.push(found);
    }
  }
  for (let i = 0; i < n; i++) {
    bars[i] = arr[i];
    yield { swap: [i], sorted: sorted.slice(), line: 7, type: 'overwrite' };
  }
  for (let i = 0; i < n; i++) sorted[i] = true;
  yield { sorted: sorted.slice(), line: 10, type: 'done' };
}

function* bucketSort(bars) {
  const n = bars.length;
  const sorted = new Array(n).fill(false);
  const maxV = Math.max(...bars.map(b => b.value));
  const minV = Math.min(...bars.map(b => b.value));
  const k = Math.max(2, Math.floor(Math.sqrt(n)));
  const buckets = Array.from({length: k}, () => []);
  yield { line: 5, sorted: sorted.slice(), type: 'idle' };
  for (let i = 0; i < n; i++) {
    const bi = Math.min(k - 1, Math.floor((bars[i].value - minV) / (maxV - minV + 1) * k));
    buckets[bi].push(bars[i]);
    yield { compare: [i], sorted: sorted.slice(), line: 7, type: 'compare' };
  }
  for (const b of buckets) b.sort((x, y) => x.value - y.value);
  const arr = [];
  for (const b of buckets) for (const x of b) arr.push(x);
  for (let i = 0; i < n; i++) {
    bars[i] = arr[i];
    yield { swap: [i], sorted: sorted.slice(), line: 13, type: 'overwrite' };
  }
  for (let i = 0; i < n; i++) sorted[i] = true;
  yield { sorted: sorted.slice(), line: 14, type: 'done' };
}

function* radixSort(bars) {
  const n = bars.length;
  const sorted = new Array(n).fill(false);
  const maxV = Math.max(...bars.map(b => b.value));
  yield { line: 2, sorted: sorted.slice(), type: 'idle' };
  for (let exp = 1; Math.floor(maxV / exp) > 0; exp *= 10) {
    const output = new Array(n);
    const count = new Array(10).fill(0);
    for (let i = 0; i < n; i++) {
      const d = Math.floor(bars[i].value / exp) % 10;
      count[d]++;
      yield { compare: [i], sorted: sorted.slice(), line: 6, type: 'compare' };
    }
    for (let i = 1; i < 10; i++) count[i] += count[i - 1];
    for (let i = n - 1; i >= 0; i--) {
      const d = Math.floor(bars[i].value / exp) % 10;
      output[--count[d]] = bars[i];
    }
    for (let i = 0; i < n; i++) {
      bars[i] = output[i];
      yield { swap: [i], sorted: sorted.slice(), line: 13, type: 'overwrite' };
    }
  }
  for (let i = 0; i < n; i++) sorted[i] = true;
  yield { sorted: sorted.slice(), line: 15, type: 'done' };
}

const CPP_KEYWORDS = ['void','int','bool','char','float','double','if','else','for','while',
  'return','break','continue','true','false','const','new','delete','auto','long','short',
  'unsigned','signed','struct','class','public','private','protected','namespace','using',
  'template','typename','static','inline','virtual','size_t','nullptr'];
const KEYWORD_RE = new RegExp('\\b(' + CPP_KEYWORDS.join('|') + ')\\b', 'g');

function highlightCpp(text) {
  if (text === '') return '&nbsp;';
  let s = text.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');
  const stash = [];
  const TOKEN = '\uE000';
  s = s.replace(/(\/\/.*$)/g, m => { stash.push('<span class="cmt">'+m+'</span>'); return TOKEN+(stash.length-1)+TOKEN; });
  s = s.replace(/("[^"]*")/g, m => { stash.push('<span class="str">'+m+'</span>'); return TOKEN+(stash.length-1)+TOKEN; });
  s = s.replace(KEYWORD_RE, '<span class="kw">$1</span>');
  s = s.replace(/\bstd::(\w+)/g, '<span class="fn">std::$1</span>');
  s = s.replace(/\b(\d+)\b/g, '<span class="num">$1</span>');
  s = s.replace(new RegExp(TOKEN+'(\\d+)'+TOKEN, 'g'), (_, i) => stash[+i]);
  return s;
}

function layout() {
  const W = barsEl.clientWidth;
  const H = barsEl.clientHeight;
  const n = bars.length;
  if (n === 0 || W === 0 || H === 0) return;
  const margin = 6;
  const usable = W - margin * 2;
  const gap = Math.max(1, Math.min(8, 80 / n));
  const barW = (usable - gap * (n - 1)) / n;
  const maxV = Math.max(...bars.map(b => b.value));
  const maxH = H;

  bars.forEach((bar, i) => {
    const x = margin + i * (barW + gap);
    bar.el.style.left = x + 'px';
    bar.el.style.width = barW + 'px';
    const h = Math.max(4, (bar.value / maxV) * maxH);
    bar.el.style.height = h + 'px';
    bar.el.textContent = n <= 30 ? bar.value : '';
  });
}

/* ============================================================
   ★★★ 核心：applyState - 单帧内完成所有视觉更新 ★★★
   顺序：清旧类 → 应用新类 → 更新位置 → 高亮代码行
   ============================================================ */
function applyState(state) {
  // 1. 清除所有视觉状态类
  bars.forEach(bar => bar.el.classList.remove('cmp', 'swap', 'piv', 'srt'));

  // 2. 应用"已排序"类（稳定的状态，长期显示）
  if (state.sorted) {
    state.sorted.forEach((isSorted, i) => {
      if (isSorted && bars[i]) bars[i].el.classList.add('srt');
    });
  }

  // 3. 应用"基准"类
  if (state.pivot !== undefined && state.pivot >= 0 && bars[state.pivot]) {
    bars[state.pivot].el.classList.add('piv');
  }

  // 4. 应用"比较"类（橙色）
  if (state.compare) {
    state.compare.forEach(i => {
      if (bars[i]) bars[i].el.classList.add('cmp');
    });
  }

  // 5. 应用"交换"类（红色，且 z-index 提升）
  if (state.swap) {
    state.swap.forEach(i => {
      if (bars[i]) bars[i].el.classList.add('swap');
    });
  }

  // 6. ★ 关键：更新位置（交换时柱子会平滑滑到新位置）
  layout();

  // 7. 高亮代码行
  if (state.line !== undefined && state.line >= 0) {
    highlightLine(state.line);
  }
}

let isDragging = false;
let startY = 0, startTopH = 0, startBotH = 0;
const MIN_H = 90;

function onPointerDown(e) {
  e.preventDefault();
  isDragging = true;
  splitter.classList.add('dragging');
  document.body.classList.add('dragging');
  startY = e.clientY;
  startTopH = vizPanel.getBoundingClientRect().height;
  startBotH = codePanel.getBoundingClientRect().height;
  try { splitter.setPointerCapture(e.pointerId); } catch (_) {}
}

function onPointerMove(e) {
  if (!isDragging) return;
  const dy = e.clientY - startY;
  let newTopH = startTopH + dy;
  let newBotH = startBotH - dy;
  if (newTopH < MIN_H) { newTopH = MIN_H; newBotH = startTopH + startBotH - MIN_H; }
  if (newBotH < MIN_H) { newBotH = MIN_H; newTopH = startTopH + startBotH - MIN_H; }
  vizPanel.style.flex = '0 0 ' + newTopH + 'px';
  codePanel.style.flex = '0 0 ' + newBotH + 'px';
  if (bars.length) {
    if (rafId) cancelAnimationFrame(rafId);
    rafId = requestAnimationFrame(layout);
  }
}

function onPointerUp(e) {
  if (!isDragging) return;
  isDragging = false;
  splitter.classList.remove('dragging');
  document.body.classList.remove('dragging');
  try { splitter.releasePointerCapture(e.pointerId); } catch (_) {}
}
