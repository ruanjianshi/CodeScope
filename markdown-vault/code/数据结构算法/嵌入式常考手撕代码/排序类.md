//冒泡排序：原理：相邻比较，大的往后冒。每轮确定一个元素的最终位置。n 个元素，确定 n-1 个，最后一个自动就位
void bubbleSort(int arr[], int n) {

  for (int i = 0; i < n - 1; i++) {
    bool swaped = false;
    for (int j = 0; j < n - i - 1; j++) {
      if (arr[j] > arr[j + 1]) {
        int tmp = arr[j + 1];
        arr[j + 1] = arr[j];
        arr[j] = tmp;
        swaped = true;
      }
    }
    if (!swaped)
      break;
  }
}
