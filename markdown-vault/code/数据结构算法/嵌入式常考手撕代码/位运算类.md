#include "main.h"

// 统计二进制中 1 的个数
// Brian Kernighan 算法（推荐）
int bit_count(unsigned int x) {
  int count = 0;

  while (x) {
    x &= (x - 1);
    count++;
  }

  return count;
}

//判断是否是2的整数次幂，2^0,2^1,2^2,..........
bool is_power_of_two(unsigned int x){
  
  return (x > 0) && ((x & (x-1)) == 0);
}

//二进制加法 = 所有位的无进位和就是 a^b  + 所有位的进位就是 a&b
unsigned int avg_no_overflow(unsigned int a, unsigned int b){
  return (a & b) + ((a ^ b) >> 1);
}

// 不用临时变量交换两数，异或的三个性质：自反，恒等，交换律
void swap_no_temp(int *a, int *b) {
    if (a != b) {          // 防止同一地址导致清零
        *a ^= *b;
        *b ^= *a;
        *a ^= *b;
    }
}

//求绝对值且不用分支：-x = ~x + 1；0xFFFF = -1,在补码中，n ^ (-1) = ~n
int abs_no_branch(int n) {
    int mask = n >> 31;            // 负数得到 -1（全1），正数得到 0
    return (n ^ mask) - mask;
}



int main() {

  std::cout << bit_count(3) << std::endl;
  std::cout << is_power_of_two(2) << std::endl;
  std::cout << avg_no_overflow(2, 4) << std::endl;
  std::cout << abs_no_branch(-2) << std::endl;

  return 0;
}
