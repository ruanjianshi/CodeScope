## Fragment: 环形缓冲实现
```c_cpp
#include <cstdint>
#include <cstddef>

struct Ringbuf{
  uint8_t buf[64];
  volatile size_t head = 0, tail = 0, cnt = 0;
};

bool rb_put(Ringbuf& rb, uint8_t b){

  if(rb.cnt == 64) return false;
  rb.buf[rb.head] = b;
  rb.head = (rb.head + 1) % 64;
  ++rb.cnt;
  return true;
}

bool rb_get(Ringbuf& rb, uint8_t& b){
  if(rb.cnt == 0) return false;
  b = rb.buf[rb.tail];
  rb.tail = (rb.tail + 1) % 64;
  --rb.cnt;
  return true;
}
```
